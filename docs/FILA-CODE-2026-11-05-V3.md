```
@'
import json
p = json.load(open(r'C:\ponte-dominio\dominio-v3.json', encoding='utf-8'))
print('versao', p['versao'], '| linhas', len(p['mensal']), '| andamento', len(p['andamento']), '| encerramento', len(p['encerramento']), '| faltou', [f['bloco'] for f in p.get('faltou') or []])
v = lambda l, c: l.get(c) or 0
for mes in ('2026-09', '2026-10'):
    m = [l for l in p['mensal'] if l['competencia'] == mes]
    q = lambda f: sum(1 for l in m if f(l))
    notas = lambda l: v(l, 'notas_saida') + v(l, 'notas_entrada')
    folha = lambda l: v(l, 'salarios') + v(l, 'decimo_terceiro') + v(l, 'pro_labore_valor')
    print(mes, '| com lancamento', q(lambda l: v(l, 'lancamentos') > 0),
          '| com a escrita fiscal', q(lambda l: v(l, 'lanc_fiscal') > 0),
          '| com lancamento e nota, sem a escrita fiscal', q(lambda l: v(l, 'lancamentos') > 0 and notas(l) > 0 and v(l, 'lanc_fiscal') == 0),
          '| com folha calculada', q(lambda l: folha(l) > 0),
          '| folha nao lancada', q(lambda l: v(l, 'lancamentos') > 0 and folha(l) > 0 and v(l, 'folha_lancada') == 0),
          '| folha lancada diferente', q(lambda l: folha(l) > 0 and v(l, 'folha_lancada') > 0 and abs(folha(l) - v(l, 'folha_lancada')) > 1),
          '| receita lancada diferente', q(lambda l: v(l, 'receita_bruta') > 0 and v(l, 'lanc_fiscal') > 0 and abs(v(l, 'receita_bruta') - v(l, 'receita_lancada')) > 1),
          '| caixa negativo', q(lambda l: v(l, 'caixa_negativo') > 0),
          '| transitoria com saldo', q(lambda l: v(l, 'transitorias') > 0),
          '| contra a natureza', q(lambda l: v(l, 'contra_natureza') > 0),
          '| sem natureza', q(lambda l: v(l, 'sem_natureza') > 0))
a = p['andamento']
print('andamento | com banco', sum(1 for x in a if v(x, 'banco_contas') > 0), '| com extrato', sum(1 for x in a if v(x, 'extrato_contas') > 0),
      '| com caixa do contador', sum(1 for x in a if x.get('caixa_contador_ate')), '| com a data do fechamento', sum(1 for x in a if x.get('fechamento_em')))
for ano in sorted({x['ano'] for x in p['encerramento']}):
    e = [x for x in p['encerramento'] if x['ano'] == ano]
    print(ano, '| empresas', len(e), '| zeradas', sum(1 for x in e if x.get('zerado_em')), '| com lancamento depois do zeramento', sum(1 for x in e if v(x, 'lancados_depois') > 0))
'@ | py -3 -
```
